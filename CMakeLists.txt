cmake_minimum_required(VERSION 3.12.0)
project(g3sc LANGUAGES C)

include(CheckCSourceCompiles)
include(CMakePushCheckState)
include(GNUInstallDirs)
include(GenerateExportHeader)
include(FindPackageHandleStandardArgs)

set(CMAKE_C_STANDARD 90)
add_compile_options(-Wno-deprecated-declarations)
add_compile_definitions(GLIB_DISABLE_DEPRECATION_WARNINGS)

option(BUILD_SHARED_LIBS "Build as a shared library" TRUE)
option(BUILD_EXAMPLES "Build examples" TRUE)

find_package(PkgConfig REQUIRED)
set(PC_DEPS_LIST glib-2.0 gobject-2.0)
pkg_check_modules(PC_DEPS REQUIRED IMPORTED_TARGET ${PC_DEPS_LIST})

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/include/g3sc.h" GMSFHDR)
string(REGEX MATCH "G3SC_VERSION_MAJOR ([0-9]*)" _ ${GMSFHDR})
set(VERSION_MAJOR ${CMAKE_MATCH_1})
string(REGEX MATCH "G3SC_VERSION_MINOR ([0-9]*)" _ ${GMSFHDR})
set(VERSION_MINOR ${CMAKE_MATCH_1})

file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
add_library(libg3sc ${SOURCE_FILES})
add_library(g3sc::libg3sc ALIAS libg3sc)
generate_export_header(libg3sc BASE_NAME "g3sc" EXPORT_FILE_NAME "g3sc-export.h")
set_property(TARGET libg3sc PROPERTY OUTPUT_NAME "g3sc")
set_property(TARGET libg3sc PROPERTY C_VISIBILITY_PRESET "hidden")
set_property(TARGET libg3sc PROPERTY POSITION_INDEPENDENT_CODE ON)
set_target_properties(libg3sc PROPERTIES VERSION ${VERSION_MAJOR}.${VERSION_MINOR} SOVERSION ${VERSION_MAJOR})
target_include_directories(libg3sc PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_include_directories(libg3sc PUBLIC "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_include_directories(libg3sc PUBLIC "${CMAKE_BINARY_DIR}")
target_link_libraries(libg3sc PUBLIC PkgConfig::PC_DEPS)

string(JOIN " " PC_DEPS_PRIVATE ${PC_DEPS_LIST})
configure_file(g3sc.pc.in g3sc.pc @ONLY)

install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/include/g3sc.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/g3sc-export.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/g3sc.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
install(TARGETS libg3sc ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

if(BUILD_EXAMPLES)
	file(GLOB_RECURSE EXAMPLE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/examples/*.c")
	foreach(EXAMPLE_FILE ${EXAMPLE_FILES})
		get_filename_component(EXAMPLE_TARGET ${EXAMPLE_FILE} NAME_WE)
        add_executable(${EXAMPLE_TARGET} ${EXAMPLE_FILE})
        target_link_libraries(${EXAMPLE_TARGET} PRIVATE g3sc::libg3sc)
	endforeach()
endif()
